function fmt(n){return '₹'+n.toLocaleString('en-IN')}
function calcNow(){
  const bal=+document.getElementById('bal').value;
  const gain=+document.getElementById('gain').value;
  document.getElementById('bal-disp').textContent=fmt(bal);
  document.getElementById('gain-disp').textContent=gain+'%';
  const profit=Math.round(bal*gain/100);
  document.getElementById('r-profit').textContent=fmt(profit);
  document.getElementById('r-payout').textContent=fmt(Math.round(profit*.8));
}
